# ------------------------ Helpers ------------------------
def _permute(bits: int, table: List[int], n_out: int) -> int:
    out = 0                                  # Will hold the permuted result
    for i, pos in enumerate(table):          # For each output bit index i, read from input position 'pos'
        if bits >> (64 - pos) & 1:           # Align so bit 'pos' of a 64-bit lane is at LSB, then mask
            out |= 1 << (n_out - 1 - i)      # Set the corresponding output bit (MSB-first numbering)
    return out                                # Return permuted integer of n_out bits (packed in Python int)

def _lrot28(x: int, s: int) -> int:
    x &= (1 << 28) - 1                        # Ensure x is 28 bits
    return ((x << s) | (x >> (28 - s))) & ((1 << 28) - 1)  # Circular left rotate by s over 28 bits

def _sbox_substitution(x48: int) -> int:
    out = 0                                   # Will accumulate eight 4-bit S-box outputs (total 32 bits)
    for i in range(8):                        # Process each 6-bit chunk for S1..S8
        block = (x48 >> (42 - 6*i)) & 0x3F    # Extract 6 bits for S-box i
        row = ((block & 0x20) >> 4) | (block & 0x01)  # Row uses outer bits b6 and b1 → 2-bit row [0..3]
        col = (block >> 1) & 0x0F             # Column uses middle 4 bits → [0..15]
        out = (out << 4) | SBOX[i][row][col]  # Append 4-bit S-box output
    return out                                # 32-bit integer of concatenated S-box outputs

def _round_f(r32: int, k48: int) -> int:
    e = _permute(r32 << 32, E, 48)            # Expand 32-bit R to 48 bits via E (shift left to align as 64-bit lane)
    s = _sbox_substitution(e ^ k48)           # XOR with subkey, then S-box substitution → 32 bits
    p = _permute(s << 32, P, 32)              # Permute with P to diffuse bits
    return p                                  # Return f(R, K)

# ------------------------ Key schedule ------------------------
def key_schedule_from_hex(hex_key: str, verbose: bool=False) -> Tuple[List[int], str]:
    if len(hex_key) != 16:                    # Expect 16 hex chars (64-bit key incl parity)
        raise ValueError("Key must be 16 hex characters (64 bits incl. parity).")
    key64 = int(hex_key, 16)                  # Parse hex key string → 64-bit integer
    k56 = _permute(key64, PC1, 56)            # Apply PC-1 to drop parity bits → 56 bits (C0||D0)
    c = (k56 >> 28) & ((1 << 28) - 1)         # Upper 28 bits → C0
    d =  k56        & ((1 << 28) - 1)         # Lower 28 bits → D0

    log = []                                  # Optional log for verbose key schedule display
    if verbose:
        log.append(f"PC1 -> 56b: C0={c:07X} D0={d:07X}")  # Show initial C0/D0 in hex

    subs: List[int] = []                      # Will hold 16 round subkeys (48-bit each)
    for rnd, s in enumerate(SHIFTS, 1):       # For rounds 1..16, rotate according to SHIFTS
        c = _lrot28(c, s)                     # Left-rotate C by s
        d = _lrot28(d, s)                     # Left-rotate D by s
        cd = (c << 28) | d                    # Concatenate to 56-bit C||D
        k48 = _permute(cd << (64 - 56), PC2, 48)  # Apply PC-2 to select 48 bits for subkey
        subs.append(k48)                      # Store subkey for this round
        if verbose:
            log.append(f"RoundKey {rnd:02d}: {k48:012X}") # Show subkey in hex
    return subs, ("\n".join(log) if verbose else "")  # Return list and optional log

# ------------------------ Block ops ------------------------
def des_block_encrypt(block8: bytes, subkeys: List[int], trace: bool=True) -> Tuple[bytes, str]:
    if len(block8) != 8:                      # DES processes exactly 8-byte blocks
        raise ValueError("DES operates on 8-byte blocks.")
    x = int.from_bytes(block8, 'big')         # Convert 8 bytes to 64-bit integer (big-endian)
    log = []                                  # Log for optional round-by-round trace

    x_ip = _permute(x, IP, 64)                # Apply initial permutation IP
    l = (x_ip >> 32) & 0xFFFFFFFF             # Split into left 32 bits L0
    r =  x_ip        & 0xFFFFFFFF             # Right 32 bits R0

    if trace:
        log.append("Encryption")              # Header
        log.append(f"After initial permutation {x_ip:016X}")  # Show IP result
        log.append(f"Round  0  {l:08X}  {r:08X}")             # Show (L0,R0)

    for i, k in enumerate(subkeys, 1):        # 16 Feistel rounds using subkeys K1..K16
        f = _round_f(r, k)                    # Compute f(R_{i-1}, K_i)
        l, r = r, (l ^ f) & 0xFFFFFFFF        # Feistel: L_i = R_{i-1}; R_i = L_{i-1} XOR f
        if trace:
            log.append(f"Round {i:2d}  {l:08X}  {r:08X}")     # Log round state (Li, Ri)

    pre_fp = (r << 32) | l                    # After round 16, swap halves (R16||L16)
    x_ct   = _permute(pre_fp, FP, 64)         # Apply final permutation FP to get ciphertext
    if trace:
        log.append(f"Cipher Text :  {x_ct:016X}")             # Log final CT
    return x_ct.to_bytes(8, 'big'), "\n".join(log)            # Return bytes and log

def des_block_decrypt(block8: bytes, subkeys: List[int], trace: bool=True) -> Tuple[bytes, str]:
    if len(block8) != 8:                      # Still requires 8 bytes
        raise ValueError("DES operates on 8-byte blocks.")
    x = int.from_bytes(block8, 'big')         # Parse 64-bit block
    log = []                                  # Log for tracing

    x_ip = _permute(x, IP, 64)                # Apply IP to ciphertext as well
    l = (x_ip >> 32) & 0xFFFFFFFF             # Split into L0
    r =  x_ip        & 0xFFFFFFFF             # and R0

    if trace:
        log.append("Decryption")              # Header
        log.append(f"After initial permutation {x_ip:016X}")  # Show IP result
        log.append(f"Round  0  {l:08X}  {r:08X}")             # Show (L0,R0)

    for i, k in enumerate(reversed(subkeys), 1):  # Use subkeys in reverse order for decryption
        f = _round_f(r, k)                    # Same f-function
        l, r = r, (l ^ f) & 0xFFFFFFFF        # Same Feistel structure
        if trace:
            log.append(f"Round {i:2d}  {l:08X}  {r:08X}")     # Log round state

    pre_fp = (r << 32) | l                    # Swap halves back
    x_pt   = _permute(pre_fp, FP, 64)         # Apply FP to recover plaintext
    if trace:
        log.append(f"Plain Text :  {x_pt:016X}")              # Log PT
    return x_pt.to_bytes(8, 'big'), "\n".join(log)            # Return bytes and log

# ------------------------ CBC / CBC-CTS (CS3 direct) ------------------------
BLOCK = 8                                    # DES block size in bytes

def _xor_bytes(a: bytes, b: bytes) -> bytes:
    return bytes(x ^ y for x, y in zip(a, b)) # Byte-wise XOR for equal-length inputs

def _split_blocks(data: bytes, size: int) -> List[bytes]:
    return [data[i:i+size] for i in range(0, len(data), size)] # Split data into chunks of size

def _bhex(b: bytes) -> str:
    return b.hex().upper()                    # Helper to format bytes as uppercase hex

def _indent(text: str, pad: str = "    ") -> str:
    return "\n".join(pad + line for line in text.splitlines()) # Indent multi-line text for pretty logs

# --- Standard CBC for full blocks (no trace, core logic preserved) ---
def des_cbc_encrypt_fullblocks(plain: bytes, subkeys: List[int], iv: bytes) -> bytes:
    assert len(iv) == BLOCK                   # IV must be 8 bytes
    out = bytearray()                         # Accumulate ciphertext blocks
    prev = iv                                 # C_0 = IV
    for blk in _split_blocks(plain, BLOCK):   # Process each 8-byte block
        assert len(blk) == BLOCK              # This path expects exact multiples (no CTS)
        x = _xor_bytes(blk, prev)             # X = P_i XOR C_{i-1}
        c, _ = des_block_encrypt(x, subkeys, trace=False)  # C_i = E(X)
        out += c                              
        prev = c                               # Update chaining value
    return bytes(out)                          # Return concatenated ciphertext

def des_cbc_decrypt_fullblocks(cipher: bytes, subkeys: List[int], iv: bytes) -> bytes:
    assert len(iv) == BLOCK                   # IV size check
    out = bytearray()                         # Accumulate plaintext
    prev = iv                                 # C_0 = IV
    for c in _split_blocks(cipher, BLOCK):    # For each ciphertext block
        assert len(c) == BLOCK                # Must be full blocks
        x, _ = des_block_decrypt(c, subkeys, trace=False)  # X = D(C_i)
        p = _xor_bytes(x, prev)               # P_i = X XOR C_{i-1}
        out += p
        prev = c                              # Update chaining value
    return bytes(out)                         # Return concatenated plaintext

# --- CBC with Ciphertext Stealing (CS3, direct two-block logic) ---
def des_cbc_encrypt_cts(plain: bytes, subkeys: List[int], iv: bytes) -> bytes:
    assert len(iv) == BLOCK                   # IV must be 8 bytes
    n = len(plain)                            # Total plaintext length
    if n == 0:
        return b""                            # Empty input -> empty output
    if 0 < n < BLOCK:
        raise ValueError("CBC-CTS (DES) needs >= 9 bytes total.")  # We require at least 9 bytes for CTS
    if n % BLOCK == 0:
        return des_cbc_encrypt_fullblocks(plain, subkeys, iv)  # Multiples of 8 -> normal CBC

    r = n % BLOCK                             # r = last partial block length (1..7)
    before_len = (n - r) - BLOCK              # Length before the final two blocks
    before = plain[:before_len]               # P1..P_{m-2} (may be empty)
    Pm1 = plain[before_len:before_len+BLOCK]  # P_{m-1}: full block
    Pm  = plain[-r:]                          # P_m: partial block (r bytes)

    C_before = des_cbc_encrypt_fullblocks(before, subkeys, iv) if before_len > 0 else b""  # CBC over earlier blocks
    C_prev = iv if len(C_before) == 0 else C_before[-BLOCK:]   # C_{m-2} = IV if none before, else last full C

    T, _ = des_block_encrypt(_xor_bytes(Pm1, C_prev), subkeys, trace=False)  # T = E(P_{m-1} XOR C_{m-2})
    Cm = T[:r]                              # C_m = first r bytes of T (stolen from T)
    Pm_prime = Pm + T[r:]                   # P_m' = P_m || tail(T) to make a full block (8 bytes)
    Cm1, _ = des_block_encrypt(_xor_bytes(Pm_prime, C_prev), subkeys, trace=False)  # C_{m-1} = E(P_m' XOR C_{m-2})
    return C_before + Cm1 + Cm              # Output order (CS3): ... || C_{m-1} || C_m

def des_cbc_decrypt_cts(cipher: bytes, subkeys: List[int], iv: bytes) -> bytes:
    assert len(iv) == BLOCK                   # IV must be 8
    n = len(cipher)                           # Total ciphertext length
    if n == 0:
        return b""                            # Empty -> empty
    if 0 < n < BLOCK:
        raise ValueError("CBC-CTS (DES) needs >= 9 bytes total.")  # Too short for CTS
    if n % BLOCK == 0:
        return des_cbc_decrypt_fullblocks(cipher, subkeys, iv)  # Multiples of 8 -> normal CBC

    r = n % BLOCK                             # r = partial tail length (1..7)
    before_len = n - BLOCK - r                # Bytes before last two (full + partial)
    C_before = cipher[:before_len]            # C1..C_{m-2} (may be empty)
    Cm1 = cipher[before_len:before_len+BLOCK] # C_{m-1}: full block
    Cm  = cipher[-r:]                         # C_m: partial tail

    P_before = des_cbc_decrypt_fullblocks(C_before, subkeys, iv) if before_len > 0 else b""  # Decrypt earlier CBC
    C_prev = iv if before_len == 0 else C_before[-BLOCK:]   # C_{m-2}

    temp, _ = des_block_decrypt(Cm1, subkeys, trace=False)  # temp = D(C_{m-1})
    Pm_prime = _xor_bytes(temp, C_prev)             # P_m' = D(C_{m-1}) XOR C_{m-2} = P_m || tail(T)
    Pm = Pm_prime[:r]                               # Recover the real partial P_m (first r bytes)
    tail = Pm_prime[r:]                             # tail(T) = remaining bytes to reconstruct T
    T = Cm + tail                                   # Rebuild T = C_m || tail(T)
    U, _ = des_block_decrypt(T, subkeys, trace=False)  # U = D(T)
    Pm1 = _xor_bytes(U, C_prev)                     # Recover P_{m-1} = U XOR C_{m-2}
    return P_before + Pm1 + Pm                      # Concatenate full plaintext

# ------------------------ VERBOSE TRACING WRAPPERS (no logic change) ------------------------
def des_cbc_encrypt_fullblocks_trace(plain: bytes, subkeys: List[int], iv: bytes) -> Tuple[bytes, str]:
    log = [f"CBC ENCRYPT (full blocks), IV={_bhex(iv)}"]  # Start log with IV
    out = bytearray()                         # Accumulate ciphertext
    prev = iv                                 # C_0 = IV
    for idx, blk in enumerate(_split_blocks(plain, BLOCK), 1):  # Enumerate blocks starting at 1
        x = _xor_bytes(blk, prev)             # X_i = P_i XOR C_{i-1}
        c, tlog = des_block_encrypt(x, subkeys, trace=True)  # Encrypt and capture DES round trace
        log.append(f"- Block {idx}")          # Block header
        log.append(f"  P{idx}   = {_bhex(blk)}")              # Show plaintext block
        log.append(f"  C{idx-1} = {_bhex(prev)}{' (IV)' if idx==1 else ''}")  # Show previous C (or IV)
        log.append(f"  X{idx}   = P{idx} XOR C{idx-1} = {_bhex(x)}")          # Show XOR result
        log.append("  DES E(X):")             # Label for inner DES trace
        log.append(_indent(tlog, "    "))     # Indent DES trace for readability
        log.append(f"  C{idx}   = {_bhex(c)}")# Show resulting ciphertext block
        out += c                              # Append to output
        prev = c                              # Update chaining value
    return bytes(out), "\n".join(log)         # Return ciphertext and combined log

def des_cbc_decrypt_fullblocks_trace(cipher: bytes, subkeys: List[int], iv: bytes) -> Tuple[bytes, str]:
    log = [f"CBC DECRYPT (full blocks), IV={_bhex(iv)}"]  # Start log with IV
    out = bytearray()                         # Accumulate plaintext blocks
    prev = iv                                 # C_0 = IV
    for idx, c in enumerate(_split_blocks(cipher, BLOCK), 1):  # Enumerate ciphertext blocks
        x, tlog = des_block_decrypt(c, subkeys, trace=True)    # Inner DES decrypt with trace
        p = _xor_bytes(x, prev)               # P_i = D(C_i) XOR C_{i-1}
        log.append(f"- Block {idx}")          # Block header
        log.append(f"  C{idx}   = {_bhex(c)}")                # Show ciphertext block
        log.append(f"  D(C{idx})= {_bhex(x)}")                # Show DES-decrypted block
        log.append(f"  C{idx-1} = {_bhex(prev)}{' (IV)' if idx==1 else ''}")  # Show previous C (or IV)
        log.append(f"  P{idx}   = D(C{idx}) XOR C{idx-1} = {_bhex(p)}")       # Show recovered plaintext block
        log.append("  DES D(C):")             # Label for DES trace
        log.append(_indent(tlog, "    "))     # Indent for readability
        out += p                              # Append plaintext
        prev = c                              # Update chaining value
    return bytes(out), "\n".join(log)         # Return plaintext and log

def des_cbc_encrypt_cts_trace(plain: bytes, subkeys: List[int], iv: bytes) -> Tuple[bytes, str]:
    n = len(plain)                            # Total message length
    if n % BLOCK == 0:                        # If multiple of 8, use full-block tracing
        ct, slog = des_cbc_encrypt_fullblocks_trace(plain, subkeys, iv)
        return ct, slog
    r = n % BLOCK                             # Partial tail length (1..7)
    log = [f"CBC-CTS ENCRYPT (CS3), IV={_bhex(iv)}, len={n}, r={r}"]  # CTS header with parameters
    before_len = (n - r) - BLOCK              # Bytes before the final two blocks
    before = plain[:before_len]               # Earlier blocks (maybe empty)
    Pm1 = plain[before_len:before_len+BLOCK]  # Penultimate full block
    Pm  = plain[-r:]                          # Final partial block

    C_before = b""                            # Will hold ciphertext of earlier blocks (if any)
    if before_len > 0:
        C_before, slog = des_cbc_encrypt_fullblocks_trace(before, subkeys, iv) # Trace earlier CBC
        log.append(slog)                      # Append trace
    C_prev = iv if len(C_before) == 0 else C_before[-BLOCK:]   # C_{m-2} for the two-block CTS step
    log.append(f"- Penultimate full block Pm-1 = {_bhex(Pm1)}") # Show P_{m-1}
    log.append(f"  C(m-2) = {_bhex(C_prev)}")                   # Show C_{m-2}

    X = _xor_bytes(Pm1, C_prev)               # X = P_{m-1} XOR C_{m-2}
    T, tlog = des_block_encrypt(X, subkeys, trace=True)         # T = E(X), with DES trace
    log.append(f"  X = Pm-1 XOR C(m-2) = {_bhex(X)}")           # Show X
    log.append("  T = E(X) DES trace:")                          # Label for DES trace
    log.append(_indent(tlog, "    "))         # Indent DES trace

    Cm = T[:r]                                # C_m = first r bytes of T (the “stolen” bytes)
    log.append(f"  Cm (partial) = first {r} bytes of T = {_bhex(Cm)}")  # Show Cm

    Pm_prime = Pm + T[r:]                     # P_m' = P_m || tail(T), a full 8-byte block
    log.append(f"  Pm' = Pm || tail(T) = {_bhex(Pm_prime)}")    # Show constructed P_m'

    X2 = _xor_bytes(Pm_prime, C_prev)         # X2 = P_m' XOR C_{m-2}
    Cm1, tlog2 = des_block_encrypt(X2, subkeys, trace=True)     # C_{m-1} = E(X2) with DES trace
    log.append(f"  X2 = Pm' XOR C(m-2) = {_bhex(X2)}")          # Show X2
    log.append("  Cm-1 = E(X2) DES trace:")                     # Label
    log.append(_indent(tlog2, "    "))       # Indent internal DES trace

    ct = C_before + Cm1 + Cm                  # Output order for CS3
    log.append(f"Ciphertext = {_bhex(ct)} (same length as plaintext)")  # Final summary
    return ct, "\n".join(log)                 # Return ciphertext and full trace

def des_cbc_decrypt_cts_trace(cipher: bytes, subkeys: List[int], iv: bytes) -> Tuple[bytes, str]:
    n = len(cipher)                           # Total ciphertext length
    if n % BLOCK == 0:                        # If multiple of 8, use full-block tracing
        pt, slog = des_cbc_decrypt_fullblocks_trace(cipher, subkeys, iv)
        return pt, slog
    r = n % BLOCK                             # Partial tail length (1..7)
    log = [f"CBC-CTS DECRYPT (CS3), IV={_bhex(iv)}, len={n}, r={r}"]  # CTS header

    before_len = n - BLOCK - r                # Bytes before last full+partial pair
    C_before = cipher[:before_len]            # C1..C_{m-2}
    Cm1 = cipher[before_len:before_len+BLOCK] # C_{m-1} full
    Cm  = cipher[-r:]                         # C_m partial
    log.append(f"- C_before bytes: {_bhex(C_before) if C_before else '(none)'}")  # Earlier ciphertext
    log.append(f"  Cm-1 (full)    = {_bhex(Cm1)}")                                # Show C_{m-1}
    log.append(f"  Cm   (partial) = {_bhex(Cm)}")                                 # Show C_m

    P_before = b""                            # Placeholder for earlier CBC plaintext
    if before_len > 0:
        P_before, slog = des_cbc_decrypt_fullblocks_trace(C_before, subkeys, iv)  # Trace earlier CBC decryption
        log.append(slog)                    # Append trace
    C_prev = iv if before_len == 0 else C_before[-BLOCK:]   # C_{m-2}
    log.append(f"  C(m-2) = {_bhex(C_prev)}")               # Show C_{m-2}

    temp, tlog1 = des_block_decrypt(Cm1, subkeys, trace=True)  # temp = D(C_{m-1}) with DES trace
    log.append("  D(Cm-1) DES trace:")                         # Label
    log.append(_indent(tlog1, "    "))                         # Indent DES trace
    Pm_prime = _xor_bytes(temp, C_prev)        # P_m' = D(C_{m-1}) XOR C_{m-2} = P_m || tail(T)
    Pm = Pm_prime[:r]                          # Extract P_m (first r bytes)
    tail = Pm_prime[r:]                        # tail(T) (remaining bytes)
    log.append(f"  Pm' = D(Cm-1) XOR C(m-2) = {_bhex(Pm_prime)}")  # Show P_m'
    log.append(f"  Pm  = first {r} bytes of Pm' = {_bhex(Pm)}")    # Show P_m
    log.append(f"  tail(T) = {_bhex(tail)}")                       # Show tail(T)

    T = Cm + tail                              # Rebuild T = C_m || tail(T)
    U, tlog2 = des_block_decrypt(T, subkeys, trace=True)  # U = D(T) with DES trace
    log.append(f"  T = Cm || tail(T) = {_bhex(T)}")        # Show T
    log.append("  D(T) DES trace:")                          # Label
    log.append(_indent(tlog2, "    "))                       # Indent
    Pm1 = _xor_bytes(U, C_prev)                # Recover P_{m-1} = U XOR C_{m-2}
    log.append(f"  Pm-1 = D(T) XOR C(m-2) = {_bhex(Pm1)}")  # Show P_{m-1}

    pt = P_before + Pm1 + Pm                   # Full plaintext concatenation
    log.append(f"Plaintext = {_bhex(pt)}")     # Final summary
    return pt, "\n".join(log)                  # Return plaintext and full trace

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Task 2 – RSA Keygen + Segmented Sign/Verify (SHA-256)
- Prompts the user to generate RSA keys (two 10-digit primes + e).
- Signs a message by hashing with SHA-256, converting to int, then
  splitting that integer into chunks strictly smaller than n (segmentation),
  and signing each chunk with RSA. (No padding – assignment allows any hash.)
- Verifies by recomputing the same chunks from the message hash and
  checking s_i^e ≡ chunk_i (mod n) for every i.
- Prints parameters and step-by-step details for screenshots.
"""

from typing import Tuple, List
from utils import gcd, modinv, powmod, is_prime_simple, sha256_int

# ======== RSA primitives ========

def rsa_keygen(p: int, q: int, e: int) -> Tuple[Tuple[int,int], Tuple[int,int], int]:
    """
    Returns (pub=(n,e), priv=(n,d), phi) after validating inputs.
    Expects p and q to be primes, e to be coprime with phi=(p-1)(q-1).
    """
    if not is_prime_simple(p) or not is_prime_simple(q):
        raise ValueError("p and q must be prime")
    n = p * q
    phi = (p - 1) * (q - 1)
    if not (1 < e < phi):
        raise ValueError("e must satisfy 1 < e < phi(n)")
    if gcd(e, phi) != 1:
        raise ValueError("e must be coprime with phi(n)")
    d = modinv(e, phi)
    return (n, e), (n, d), phi

def rsa_encrypt_int(m: int, pub: Tuple[int,int]) -> int:
    """ RSA encryption on a single integer representative (m must be in [0, n-1]). """
    n, e = pub
    if m < 0 or m >= n:
        raise ValueError("Message integer out of range for RSA modulus")
    return powmod(m, e, n)

def rsa_decrypt_int(c: int, priv: Tuple[int,int]) -> int:
    """ RSA decryption on a single integer representative. """
    n, d = priv
    return powmod(c, d, n)

# ======== Segmentation helpers (for large hash integers) ========

def _segment_int_for_n(x: int, n: int) -> List[int]:
    """
    Split integer x into big-endian chunks that are each < n, so
    each chunk can be signed safely: s_i = chunk_i^d mod n.
    We choose chunk size = (bitlen(n) - 1) to guarantee chunk < n.
    """
    if n <= 1:
        raise ValueError("Invalid modulus n")
    kbits = max(1, n.bit_length() - 1)    # strictly less than n
    mask  = (1 << kbits) - 1
    parts: List[int] = []
    y = x
    if y == 0:
        parts.append(0)
    else:
        stack = []
        while y > 0:
            stack.append(y & mask)
            y >>= kbits
        parts = list(reversed(stack))      # big-endian order
    return parts

def rsa_sign_hash(h: int, priv: Tuple[int, int]) -> List[int]:
    """
    Sign an integer hash by segmenting into <n chunks and applying RSA per-chunk.
    Returns list of signature integers [s1, s2, ...].
    """
    n, d = priv
    chunks = _segment_int_for_n(h, n)
    return [powmod(c, d, n) for c in chunks]

def rsa_verify_hash(h: int, sigs: List[int], pub: Tuple[int, int]) -> bool:
    """
    Verify segmented signature: recompute chunks from hash and check s_i^e == chunk_i (mod n).
    """
    n, e = pub
    chunks = _segment_int_for_n(h, n)
    if len(chunks) != len(sigs):
        return False
    for c, s in zip(chunks, sigs):
        if powmod(s, e, n) != c:
            return False
    return True

def hash_message_to_int(msg: str) -> int:
    """ Hash a UTF-8 message with SHA-256 and return big-endian integer. """
    return sha256_int(msg)

# ======== Pretty-print helpers ========

def _hex(x: int) -> str:
    return hex(x)[2:].upper()

def _chunks_hex(chunks: List[int]) -> str:
    return "[" + ", ".join("0x" + _hex(c) for c in chunks) + "]"

def _sig_to_hexlist(sigs: List[int], n: int) -> str:
    # width: number of hex nibbles to display uniformly
    width = (n.bit_length() + 3) // 4
    return "-".join(f"{s:0{width}X}" for s in sigs)

def _sig_from_hexlist(s: str) -> List[int]:
    toks = [t for t in s.replace(",", "-").split("-") if t.strip()]
    return [int(t.strip(), 16) for t in toks]

def _is_10_digit(n: int) -> bool:
    return 1_000_000_000 <= n <= 9_999_999_999

def short_message_attack(n: int, e: int, m: int, max_guess: int = 1000):
    """Return (ciphertext, recovered_or_None)."""
    c = powmod(m, e, n)
    found = None
    for guess in range(max_guess):
        if powmod(guess, e, n) == c:
            found = guess
            break
    return c, found

"""

- Verbose tracing for report/screenshots

def rsa_encrypt_int(m: int, n: int, e: int, trace: bool = True) -> int:
    """Textbook RSA encryption with optional trace."""
    if not (0 <= m < n):
        raise ValueError(f"Message must satisfy 0 <= m < n (got m={m}, n={n})")
    if trace:
        print(f"\n[Encrypt] m={m}, n={n}, e={e}")
        print(f"  Compute m^e = {m}^{e} = {m**e}")
        print(f"  Reduce modulo n: c = m^e mod n = {(m**e) % n}")
    c = powmod(m, e, n)
    if trace:
        reduced = (m**e >= n)
        print(f"  powmod result: c = {c}  ({'with' if reduced else 'no'} modular reduction)")
    return c

def brute_force_short_message(n: int, e: int, c: int, trace: True, peek: int = 30):
    """
    Naive brute force: test m' in [0..n-1] until powmod(m', e, n) == c.
    Prints the first 'peek' tries, then announces the hit.
    Returns the recovered m (or None if not found).
    """
    print(f"\n[Attack: Brute Force] Find m' s.t. (m')^e mod n == c ({c})")
    for guess in range(n):
        val = powmod(guess, e, n)
        if guess < peek:
            print(f"  try m'={guess:3d} -> (m')^e mod n = {val}")
        if val == c:
            print(f"  ==> Found match: m = {guess}")
            return guess
    print("  ==> No match in searched range (unexpected for two-digit n).")
    return None

    pt, _ = des_block_decrypt(block8, subkeys, trace=False)
    return pt

# ---------- ECB + CTS over arbitrary bytes (new here) ----------
def ecb_cts_encrypt_bytes(plain: bytes, subkeys: List[int]) -> bytes:
    n = len(plain)
    if n == 0:
        return b""
    if n < BLOCK:
        raise ValueError("ECB-CTS needs at least one full block (>= 9 bytes to have a partial tail).")
    if n % BLOCK == 0:
        out = bytearray()
        for blk in _split_blocks(plain, BLOCK):
            out += _des_ecb_encrypt_block(blk, subkeys)
        return bytes(out)

    r = n % BLOCK
    before_len = (n - r) - BLOCK
    before = plain[:before_len]
    Pm1 = plain[before_len:before_len+BLOCK]
    Pm  = plain[-r:]

    C_before = bytearray()
    for blk in _split_blocks(before, BLOCK):
        if not blk:
            continue
        C_before += _des_ecb_encrypt_block(blk, subkeys)

    T = _des_ecb_encrypt_block(Pm1, subkeys)
    Cm = T[:r]
    Pm_prime = Pm + T[r:]
    Cm1 = _des_ecb_encrypt_block(Pm_prime, subkeys)

    return bytes(C_before) + Cm1 + Cm

def ecb_cts_decrypt_bytes(cipher: bytes, subkeys: List[int]) -> bytes:
    n = len(cipher)
    if n == 0:
        return b""
    if n < BLOCK:
        raise ValueError("ECB-CTS needs at least one full block.")
    if n % BLOCK == 0:
        out = bytearray()
        for c in _split_blocks(cipher, BLOCK):
            out += _des_ecb_decrypt_block(c, subkeys)
        return bytes(out)

    r = n % BLOCK
    before_len = n - BLOCK - r
    C_before = cipher[:before_len]
    Cm1 = cipher[before_len:before_len+BLOCK]
    Cm  = cipher[-r:]

    P_before = bytearray()
    for c in _split_blocks(C_before, BLOCK):
        if not c:
            continue
        P_before += _des_ecb_decrypt_block(c, subkeys)

    Pm_prime = _des_ecb_decrypt_block(Cm1, subkeys)  # = Pm || tail(T)
    Pm  = Pm_prime[:r]
    tail = Pm_prime[r:]
    T = Cm + tail
    Pm1 = _des_ecb_decrypt_block(T, subkeys)

    return bytes(P_before) + Pm1 + Pm

#!/usr/bin/env python3
# -*- coding: utf-8 -*-
"""
Shared utilities: number theory, hex helpers, and simple wrappers.
"""

import math
import hashlib
from typing import Tuple, List

# ================= Number theory / math =================

def egcd(a: int, b: int) -> Tuple[int, int, int]:
    if b == 0:
        return (a, 1, 0)
    g, x1, y1 = egcd(b, a % b)
    return (g, y1, x1 - (a // b) * y1)

def gcd(a: int, b: int) -> int:
    while b:
        a, b = b, a % b
    return abs(a)

def modinv(a: int, n: int) -> int:
    g, x, _ = egcd(a, n)
    if g != 1:
        raise ValueError("No modular inverse")
    return x % n

def powmod(a: int, e: int, n: int) -> int:
    result = 1
    a %= n
    for bit in bin(e)[2:]:
        result = (result * result) % n
        if bit == '1':
            result = (result * a) % n
    return result

def is_prime_simple(n: int) -> bool:
    if n < 2:
        return False
    if n == 2:
        return True
    if n % 2 == 0:
        return False
    max_divisor = int(math.isqrt(n)) + 1
    for i in range(3, max_divisor, 2):
        if n % i == 0:
            return False
    return True

# ================= Hash helper =================

def sha256_int(msg: str) -> int:
    return int.from_bytes(hashlib.sha256(msg.encode('utf-8')).digest(), 'big')

# ================= Bytes/hex helpers =================

def hex_to_bytes(s: str) -> bytes:
    if s is None:
        raise ValueError("Empty hex string")
    s = s.strip().lower()
    for ch in (" ", ":", "-"):
        s = s.replace(ch, "")
    if s.startswith("0x"):
        s = s[2:]
    if len(s) % 2 == 1:
        s = "0" + s
    try:
        return bytes.fromhex(s)
    except ValueError:
        raise ValueError(f"Invalid hex string: {s}")

def bytes_to_hex(b: bytes) -> str:
    return b.hex().upper()

MAIN_MENU = """\
***** LOCK && KEY *****
1. Confidentiality Only (DES from scratch; CBC + CTS)
2. Authentication Only (RSA signatures)
3. Confidentiality & Authentication (combine 1 + 2)
4. RSA cryptanalysis (short-message brute force)
5. Image encryption (ECB/CBC with CTS)
0. Exit
************************
"""

def main():
    # Shared RSA state (so option 2 and 3 can reuse generated keys)
    state = {"pub": None, "priv": None, "last_msg": None, "last_sig": None}

    while True:
        print(MAIN_MENU)
        ch = input("> ").strip()
        try:
            if ch == "1":
                # Reuse Task-1's own interactive flow with tracing
                action_confidentiality_only_verbose()
            elif ch == "2":
                menu_authentication_only(state)
            elif ch == "3":
                menu_conf_and_auth(state)
            elif ch == "4":
                menu_rsa_cryptanalysis()
            elif ch == "5":
                menu_images()
            elif ch == "0":
                break
            else:
                print("Invalid option.")
        except KeyboardInterrupt:
            print("\nInterrupted.")
        except Exception as e:
            print("Error:", e)

if __name__ == "__main__":
    main()
